// PANTALLA PRINCIPAL
// ==========================================

class Inicio extends StatefulWidget {
  final bool medico;

  const Inicio({super.key, required this.medico});

  @override
  State<Inicio> createState() => _InicioState();
}

class _InicioState extends State<Inicio> {
  int pagina = 0;

  late final Esp32Service esp32;
  Timer? temporizador;

  bool conectado = false;
  bool cargando = false;

  String estadoConexion = 'Esperando conexión con ESP32';

  Map<String, dynamic>? datosSensor;

  final List<double> historialPsi = [];
  final List<double> historialRaw = [];

  double maximoSesion = 0;
  double fuerzaMaximaSesion = 0;

  @override
  void initState() {
    super.initState();

    esp32 = Esp32Service(baseUrl: 'http://192.168.4.1');
    iniciarActualizacion();
  }

  Future<void> consultarESP32() async {
    if (cargando) return;

    if (mounted) {
      setState(() => cargando = true);
    }

    try {
      final datos = await esp32.getJson('/api/status');

      if (!mounted) return;

      setState(() {
        datosSensor = datos;
        conectado = true;
        estadoConexion = 'ESP32 conectado';

        final psi = datos['pressurePsi'];
        final raw = datos['deltaRaw'];
        final fuerza = datos['forceN'];

        if (psi is num && psi.isFinite) {
          final lecturaPsi = psi.toDouble();

          historialPsi.add(lecturaPsi);

          if (historialPsi.length > 30) {
            historialPsi.removeAt(0);
          }

          if (lecturaPsi.abs() > maximoSesion) {
            maximoSesion = lecturaPsi.abs();
          }
        }

        if (raw is num && raw.isFinite) {
          historialRaw.add(raw.toDouble());

          if (historialRaw.length > 30) {
            historialRaw.removeAt(0);
          }
        }

        if (fuerza is num && fuerza.isFinite) {
          if (fuerza.abs() > fuerzaMaximaSesion) {
            fuerzaMaximaSesion = fuerza.abs().toDouble();
          }
        }
      });
    } catch (_) {
      if (!mounted) return;

      setState(() {
        conectado = false;
        estadoConexion = 'ESP32 desconectado';
      });
    } finally {
      if (mounted) {
        setState(() => cargando = false);
      }
    }
  }

  void iniciarActualizacion() {
    consultarESP32();

    temporizador = Timer.periodic(
      const Duration(seconds: 2),
      (_) => consultarESP32(),
    );
  }

  @override
  void dispose() {
    temporizador?.cancel();
    esp32.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final titulos = ['Inicio', 'Alertas', 'Análisis'];

    final pantallas = [
      widget.medico
          ? PanelMedico(
              conectado: conectado,
              estadoConexion: estadoConexion,
              datosSensor: datosSensor,
              historialPsi: historialPsi,
              historialRaw: historialRaw,
              maximoSesion: maximoSesion,
              fuerzaMaximaSesion: fuerzaMaximaSesion,
              abrirAnalisis: () => setState(() => pagina = 2),
            )
          : PanelPaciente(
              conectado: conectado,
              estadoConexion: estadoConexion,
              datosSensor: datosSensor,
            ),
      Alertas(conectado: conectado, datosSensor: datosSensor),
      Analisis(
        historialPsi: historialPsi,
        historialRaw: historialRaw,
        conectado: conectado,
        maximoSesion: maximoSesion,
        fuerzaMaximaSesion: fuerzaMaximaSesion,
      ),
    ];

    return Scaffold(
      appBar: AppBar(
        backgroundColor: azul,
        foregroundColor: Colors.white,
        title: Text('PECTUSCARE - ${titulos[pagina]}'),
        actions: [
          IconButton(
            icon: const Icon(Icons.logout),
            onPressed: () {
              Navigator.pushReplacement(
                context,
                MaterialPageRoute(builder: (_) => const Roles()),
              );
            },
          ),
        ],
      ),
      body: pantallas[pagina],
      bottomNavigationBar: NavigationBar(
        selectedIndex: pagina,
        onDestinationSelected: (i) => setState(() => pagina = i),
        destinations: const [
          NavigationDestination(icon: Icon(Icons.home), label: 'Inicio'),
          NavigationDestination(
            icon: Icon(Icons.notifications),
            label: 'Alertas',
          ),
          NavigationDestination(
            icon: Icon(Icons.show_chart),
            label: 'Análisis',
          ),
        ],
      ),
    );
  }
}

class Analisis extends StatelessWidget {
  final List<double> historialPsi;
  final List<double> historialRaw;
  final bool conectado;
  final double maximoSesion;
  final double fuerzaMaximaSesion;

  const Analisis({
    super.key,
    required this.historialPsi,
    required this.historialRaw,
    required this.conectado,
    required this.maximoSesion,
    required this.fuerzaMaximaSesion,
  });

  @override
  Widget build(BuildContext context) {
    final hayDatos = historialPsi.isNotEmpty;
    final actual = hayDatos ? historialPsi.last : 0.0;

    final promedio = hayDatos
        ? historialPsi.map((v) => v.abs()).reduce((a, b) => a + b) /
            historialPsi.length
        : 0.0;

    final proporcion =
        maximoSesion > 0 ? actual.abs() / maximoSesion : 0.0;

    String intensidad;
    Color colorIntensidad;
    double barra;

    if (!hayDatos || maximoSesion == 0) {
      intensidad = 'Sin datos';
      colorIntensidad = Colors.grey;
      barra = 0;
    } else if (proporcion < 0.33) {
      intensidad = 'Baja';
      colorIntensidad = Colors.green;
      barra = proporcion;
    } else if (proporcion < 0.66) {
      intensidad = 'Media';
      colorIntensidad = Colors.orange;
      barra = proporcion;
    } else {
      intensidad = 'Alta';
      colorIntensidad = Colors.redAccent;
      barra = proporcion;
    }

    final ultimasPsi = historialPsi.reversed.take(5).toList();

    return ListView(
      padding: const EdgeInsets.all(18),
      children: [
        const Text(
          'Análisis de presión',
          style: TextStyle(
            fontSize: 24,
            fontWeight: FontWeight.bold,
            color: Color(0xFF173447),
          ),
        ),
        const SizedBox(height: 5),
        const Text(
          'Monitoreo experimental en tiempo real',
          style: TextStyle(color: Colors.grey),
        ),
        const SizedBox(height: 18),
        tarjeta(
          child: Row(
            children: [

// ==========================================
// GRAFICA
// ==========================================

class Grafica extends CustomPainter {
  final List<double> valores;

  Grafica(this.valores);

  @override
  void paint(Canvas canvas, Size size) {
    if (valores.length < 2) return;

    const margen = 10.0;
    final ancho = size.width;
    final alto = size.height;

    final cuadricula = Paint()
      ..color = const Color(0xFF496174).withValues(alpha: 0.65)
      ..strokeWidth = 0.8;

    for (int i = 0; i <= 4; i++) {
      final y = alto * i / 4;
      canvas.drawLine(Offset(0, y), Offset(ancho, y), cuadricula);
    }

    final puntos = <Offset>[];

    final minimo = valores.reduce((a, b) => a < b ? a : b);
    final maximo = valores.reduce((a, b) => a > b ? a : b);
    final rango = (maximo - minimo).abs();

    for (int i = 0; i < valores.length; i++) {
      final x =
          margen + (ancho - margen * 2) * i / (valores.length - 1);

      final normalizado =
          rango > 0 ? (valores[i] - minimo) / rango : 0.5;

      final y = alto - margen -
          normalizado * (alto - margen * 2);

      puntos.add(Offset(x, y));
    }

    final path = Path()..moveTo(puntos.first.dx, puntos.first.dy);

    for (int i = 1; i < puntos.length; i++) {
      final anterior = puntos[i - 1];
      final actual = puntos[i];
      final medioX = (anterior.dx + actual.dx) / 2;

      path.quadraticBezierTo(
        medioX,
        anterior.dy,
        actual.dx,
        actual.dy,
      );
    }

    final linea = Paint()
      ..color = turquesa
      ..strokeWidth = 2.5
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    canvas.drawPath(path, linea);

    final brillo = Paint()
      ..color = turquesa.withValues(alpha: 0.20)
      ..style = PaintingStyle.fill;

    canvas.drawCircle(puntos.last, 8, brillo);

    final puntoActual = Paint()
      ..color = turquesa
      ..style = PaintingStyle.fill;

    canvas.drawCircle(puntos.last, 4, puntoActual);
  }

  @override
  bool shouldRepaint(covariant Grafica oldDelegate) => true;
}
